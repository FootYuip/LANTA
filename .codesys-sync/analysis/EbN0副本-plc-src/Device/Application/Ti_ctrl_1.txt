# kind=program
# name=Ti_ctrl_1
# objectId=e86aacd1-719d-4a98-a0f2-ae7fae70687c

VAR
    Axis3_Ctrl : Ti_Ctrl_FB;
    bRemote2LocalStop : BOOL;
    bLocal2RemoteStop : BOOL;
    Error : INT;
    Axis3Limit_pos : REAL;
    Axis3Limit_Neg : REAL;
    bRTriger : Standard.R_TRIG;
    bFTriger : Standard.F_TRIG;
    Stop3_Count : INT;
    bStop_3 : BOOL;
    rPosModeStepBias : LREAL;
    rPosModeStep : LREAL;
    rStepBiasMax : LREAL;
    rStepBias : LREAL;
    rStep : LREAL;
    rAuToStepBias : LREAL;
    rAutoStep : LREAL;
    TbAutoTrigger : Standard.F_TRIG;
    RbAutoTrigger : Standard.R_TRIG;
    rAutoTrackingBiasMax : LREAL;
    rLocalInitPosModeSpeed : REAL;
    rRemoteInitPosModeSpeed : REAL;
    iCountOffAuto : INT;
    iCasePBAutoTrack : INT;
    rKAuToStepBias : LREAL;
    bVisError : BOOL;
END_VAR

// --- implementation (ST) ---
IF GVL.Error_code <> 0 THEN
	bVisError := TRUE;
ELSE
	bVisError := FALSE;
END_IF
(*
//远控轴参量赋值
Axis2Remote.iMode:=Modbus_Slave.iMode_2;
Axis2Remote.rVelCmd:=Modbus_Slave.rVelcmd_2;
Axis2Remote.rPosCmd:=Modbus_Slave.rPosCmd_2;
Axis2Remote.rP:=Modbus_Slave.rP_2;
Axis2Remote.rI:=Modbus_Slave.rI_2;
Axis2Remote.rD:=Modbus_Slave.rD_2;
Axis2Remote.rPIDThre:=Modbus_Slave.rThreshold_2;
Axis2Remote.rAcceleration:=Modbus_Slave.rAcceleration_2;
Axis2Remote.rDeceleration:=Modbus_Slave.rDeceleration_2;
Axis2Remote.rVel_Limit:=Modbus_Slave.rVelLim_2;
*)
// 倾斜轴状态写入
Axis3Status.Error := GVL.bTiStsBit3;
Axis3Status.ErrorID := GVL.eTiErrorcode;
Axis3Status.bServo_Enable := GVL.bTiStsBit0 AND GVL.bTiStsBit1 AND GVL.bTiStsBit2;
Axis3Status.bBrake := NOT (GVL.bTiStsBit0 AND GVL.bTiStsBit1 AND GVL.bTiStsBit2);
Axis3Status.Disabled := NOT (GVL.bTiStsBit0 AND GVL.bTiStsBit1 AND GVL.bTiStsBit2);
Axis3Remote.rSoft_limit_Negative:=-190;Axis3Remote.rSoft_limit_Postive:=190;
Axis3Local.rSoft_limit_Negative:=-190;Axis3Local.rSoft_limit_Postive:=190;

//速度的写入
rFctVelReal_3:=DINT_TO_REAL(diTiActVel)/524288.0*360.0;
//位置的写入，方位0对应编码器0，所以实际方位是：-1*编码器值
//rFctPosReal_3:=DINT_TO_REAL(diTiActPos)/524288.0*360.0-11.413-0.08;
rFctPosReal_3:=DINT_TO_REAL(diTiActPos)/524288.0*360.0;
//软限位控制
Axis3Limit_pos:=190;
Axis3Limit_Neg:=-190;
IF Axis3Remote.rPosCmd>Axis3Limit_pos THEN
	Axis3Remote.rPosCmd:=190;
ELSIF Axis3Remote.rPosCmd<Axis3Limit_Neg THEN
	 Axis3Remote.rPosCmd:=-190;  
END_IF

IF Axis3Local.rPosCmd>Axis3Limit_pos THEN
	Axis2Local.rPosCmd:=190;
ELSIF Axis3Local.rPosCmd<Axis3Limit_Neg THEN
	 Axis3Local.rPosCmd:=-190;  
END_IF

//停止标志符号判断
IF  ABS(rFctVelReal_3)<0.035 THEN
	Stop3_Count:=Stop3_Count+1;
	IF Stop3_Count>100 THEN
		Stop3_Count:=101;
		bStop_3:=TRUE;;
	END_IF
ELSE
	Stop3_Count:=0;
	bStop_3:=FALSE;
END_IF
Axis3Status.Moving := NOT bStop_3;
//PID使能赋值
IF Axis3Remote.iMode=2 OR Axis3Remote.iMode=4 THEN
	Axis3Remote.bPIDEnable:=TRUE;
ELSE
	Axis3Remote.bPIDEnable:=FALSE;
END_IF

//本远控逻辑保护
//远控切本控逻辑保护
bRTriger(CLK:=GVL.bLocalCtrl);
bFTriger(CLK:=GVL.bLocalCtrl);

IF Axis3Status.bServo_Enable AND (bRTriger.Q ) THEN//如果速度不为0且本控发生切换，则减速停止标识为True
	bRemote2LocalStop:=TRUE;
END_IF
IF bRemote2LocalStop  THEN //若减速停止标识符为True 
	
	IF  bStop_3=TRUE THEN//且电机速度不为0 开始减速 减速完成后下使能，下使能完成后复位标识符复位
		Axis3Local.iMode:=0;
		Axis3Remote.iMode:=0;
		IF Axis3Status.bServo_Enable=FALSE THEN
			bRemote2LocalStop:=FALSE;
		END_IF
	ELSE
		Axis3Remote.iMode:=7;
	END_IF
END_IF

//本控切远控
IF Axis3Status.bServo_Enable AND (bFTriger.Q ) THEN//如果速度不为0且本控发生切换，则减速停止标识为True
	bLocal2RemoteStop:=TRUE;
END_IF
IF bLocal2RemoteStop  THEN //若减速停止标识符为True 
	
	IF  bStop_3=TRUE THEN//且电机速度不为0 开始减速 减速完成后下使能，下使能完成后复位标识符复位
		Axis3Local.iMode:=0;
		Axis3Remote.iMode:=0;
		IF Axis3Status.bServo_Enable=FALSE THEN
			bLocal2RemoteStop:=FALSE;
		END_IF
	ELSE
		Axis3Remote.iMode:=7;
		Axis3Local.iMode:=7;
	END_IF
END_IF
(*
//自跟踪
IF GVL.bStartTrack THEN
	Axis2Remote.bStartAutoTrack := TRUE;
	Axis2Remote.rChanU := -AutoTrackingCal.rChanUa;
ELSE
	Axis2Remote.bStartAutoTrack := FALSE;	
	Axis2Remote.rChanU := 0;
END_IF
//拉偏
Axis2Remote.rPIDCmdPos:=RealTimePosCal.rInterAzPos+Modbus_Slave.rBias_2;
*)
//限位
IF Axis3Remote.rPIDCmdPos<-190 THEN
	Axis3Remote.rPIDCmdPos:=-190;
ELSIF Axis3Remote.rPIDCmdPos>190 THEN
	Axis3Remote.rPIDCmdPos:=190;
END_IF

Axis3_Ctrl(sLocal:=Axis3Local, sCtrl:=Axis3Remote, sStatus:=Axis3Status, sVisible:=Axis3Visible, rRealPos:=rFctPosReal_3, rPosLimK:=0.34);
