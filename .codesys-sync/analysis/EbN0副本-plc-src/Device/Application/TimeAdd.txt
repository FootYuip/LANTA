VAR
    icount : INT;
    iSecondLast : INT;
END_VAR

// --- implementation (ST) ---
// 监测时统串口状态，时间插值
IF iSecondLast <> GPRMC_1.iSecond THEN
	icount := 0;
ELSE
	icount := icount + 1;
END_IF
iSecondLast := GPRMC_1.iSecond;

IF icount > 2000 THEN
	icount := 0;
END_IF
