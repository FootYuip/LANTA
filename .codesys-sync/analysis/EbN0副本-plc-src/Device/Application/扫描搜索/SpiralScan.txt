# kind=functionBlock
# name=SpiralScan
# objectId=6fc58782-b739-42b8-933a-169e9b52e654

VAR_INPUT
    enable : BOOL;
    delta : REAL;
    delta1 : REAL;
    n_steps : REAL;
    agc : REAL;
    locked : BOOL;
    AzError : REAL;
    ELError : REAL;
    bTrackState : BOOL;
END_VAR

VAR_OUTPUT
    azimuth : REAL;
    elevation : REAL;
    done : BOOL;
    bType : BOOL;
END_VAR

VAR
    lock_count : INT;
    best_agc : REAL;
    best_az : REAL;
    best_el : REAL;
    bInit : BOOL;
    icount0 : INT;
    icount1 : INT;
    x : REAL;
    y : REAL;
    dx : REAL;
    dy : REAL;
    step_len : REAL;
    step_count : REAL;
    segment_progress : REAL;
    move_count : REAL;
    state : INT;
    az0 : REAL;
    el0 : REAL;
    az1 : REAL;
    el1 : REAL;
    segment_az_step : REAL;
    segment_el_step : REAL;
    side_repeat : REAL;
    icountInit : INT;
    iSearchMax : INT;
    iCountSearch : INT;
    rTemp : REAL;
END_VAR

// --- implementation (ST) ---
IF NOT enable THEN
    state := 0;
    azimuth := 0.0;
    elevation := 0.0;
    done := FALSE;
	lock_count := 0;
	icount0 := 0;
	segment_progress := 0.0;
	bInit := FALSE;
	best_agc := 0.0;
    best_az := 0.0;
    best_el := 0.0;
	icountInit := 0;
	iCountSearch := 0;
	bType := FALSE;
ELSE
    CASE state OF
        0:
			IF bTrackState THEN
				azimuth := 0.0;
				elevation := 0.0;
				done := TRUE;
				state := 3;
			ELSE
				state := 1;
				lock_count := 0;
			END_IF
        1:
            IF bTrackState THEN
				azimuth := 0.0;
				elevation := 0.0;
				done := TRUE;
				state := 3;
			ELSIF NOT bInit THEN
					x := 0.0; y := 0.0;
					dx := 1.0; dy := 0.0;
					step_len := 1.0;
					step_count := 0.0;
					move_count := 1.0;
					side_repeat := 0.0;
					az0 := 0.0; el0 := 0.0;
					az1 := delta * x;
					el1 := delta * y;
					segment_progress := 0.0;
					bInit := TRUE;
					iSearchMax := TO_INT(delta / n_steps);
					IF ABS(delta) < 1E-5 THEN
						rTemp := 0;
					ELSE
						rTemp := delta1 / delta;
					END_IF
			ELSE
					iCountSearch := iCountSearch + 1;
					IF iCountSearch <= iSearchMax THEN
						azimuth := n_steps * iCountSearch;
						elevation := n_steps * iCountSearch * rTemp;
					ELSIF iCountSearch <= 2 * iSearchMax THEN
						azimuth := delta - n_steps * (iCountSearch - iSearchMax);
						elevation := (delta - n_steps * (iCountSearch - iSearchMax)) * rTemp;
					ELSIF iCountSearch <= 3 * iSearchMax THEN
						azimuth := -1*n_steps * (iCountSearch - 2*iSearchMax);
						elevation := (-1*n_steps * (iCountSearch - 2*iSearchMax)) * rTemp;
					ELSIF iCountSearch < 4 * iSearchMax THEN
						azimuth := n_steps * (iCountSearch - 3*iSearchMax) - delta;
						elevation := (n_steps * (iCountSearch - 3*iSearchMax) - delta) * rTemp;
					END_IF

					IF ABS(delta1) < 1E-4 THEN
						elevation := 0;
					END_IF

					IF iCountSearch >= 4 * iSearchMax THEN
						azimuth := 0.0;
						elevation := 0.0;
						done := TRUE;
						state := 3;
					ELSE
						IF locked THEN
						   lock_count := lock_count + 1;
						ELSE
						   lock_count := 0;
						END_IF
						IF lock_count >= 10 THEN
						   azimuth := -Azerror;
						   elevation := -elError;
						   done := TRUE;
						   state := 2;
						END_IF
					END_IF
			END_IF
        2:
            done := TRUE;
        3:
            azimuth := 0.0;
            elevation := 0.0;
            done := TRUE;
    END_CASE

	IF state = 2 THEN
		bType := TRUE;
	ELSIF state = 3 THEN
		bType := FALSE;
	END_IF
END_IF;
