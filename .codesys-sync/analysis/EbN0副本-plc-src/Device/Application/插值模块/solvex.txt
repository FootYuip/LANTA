// 计算3x3矩阵的伴随矩阵
Determinant := M[1, 1] * (M[2, 2] * M[3, 3] - M[2, 3] * M[3, 2]) -
               M[1, 2] * (M[2, 1] * M[3, 3] - M[2, 3] * M[3, 1]) +
               M[1, 3] * (M[2, 1] * M[3, 2] - M[2, 2] * M[3, 1]);

IF ABS(Determinant) > 1.0E-6 THEN
    FOR i := 1 TO 3 DO
        FOR j := 1 TO 3 DO
            AdjointMatrix[j, i] := ((M[(i MOD 3) + 1, (j MOD 3) + 1] * M[((i + 1) MOD 3) + 1, ((j + 1) MOD 3) + 1] -
                                   M[(i MOD 3) + 1, ((j + 1) MOD 3) + 1] * M[((i + 1) MOD 3) + 1, (j MOD 3) + 1]) / Determinant);
        END_FOR
    END_FOR
END_IF

// 计算X = AdjointMatrix * G
FOR i := 1 TO 3 DO
    X[i] := 0.0;
    FOR j := 1 TO 3 DO
        X[i] := X[i] + AdjointMatrix[i, j] * G[j];
    END_FOR
END_FOR
