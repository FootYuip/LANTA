VAR_INPUT
    X : ARRAY[1..5] OF LREAL;
    Y : ARRAY[1..5] OF LREAL;
    y0 : REAL;
    yn : REAL;
END_VAR

VAR_OUTPUT
    M : ARRAY[1..5] OF LREAL;
    A : ARRAY[1..5] OF LREAL;
    D : ARRAY[1..3] OF LREAL;
    g : ARRAY[1..3] OF LREAL;
    h : ARRAY[1..4] OF LREAL;
END_VAR

VAR
    n : INT;
    ypie : ARRAY[1..5] OF REAL;
    j : INT;
    i : INT;
    u : ARRAY[1..3] OF REAL;
    n_ : ARRAY[1..3] OF REAL;
    M_ : ARRAY[1..3] OF REAL;
    solve : solvex;
    iCounttest : INT;
    miu : ARRAY[1..4] OF REAL;
END_VAR

// --- implementation (ST) ---
//此函数为M值求值函数
//D,h,A,g,M输出量分别为系数矩阵D，插值宽度h，差商表A，g值,M值 
n:=5;
//A(:,1)=Y'
A[1,1]:=Y[1];A[2,1]:=Y[2];A[3,1]:=Y[3];A[4,1]:=Y[4];A[5,1]:=Y[5];

//A(:,1)=Y'
FOR j :=2 TO 5 DO
	FOR i:=j TO 5 DO
		A[i,j]:=(A[i,j-1]- A[i-1,j-1])/(X[i]-X[i-j+1]);
	END_FOR
END_FOR


FOR i:=1 TO (n-1) DO
	h[i]:=X[i+1]-X[i];
END_FOR


//h[1]:=X[2]-X[1];h[2]:=X[3]-X[2];h[3]:=X[4]-X[3];h[4]:=X[5]-X[4];



//D[1,1]:=2;D[2,2]:=2;D[3,3]:=2;

FOR i:=1 TO n-2 DO
	D[i,i]:=2; 
	IF i=1 THEN
		g[i]:=(6/(h[i+1]+h[i]))*(A[i+2,2]-A[i+1,2])-h[i]/(h[i]+h[i+1])*y0;
	ELSIF i=(n-2) THEN
		g[i]:=(6/(h[i+1]+h[i]))*(A[i+2,2]-A[i+1,2])-(1-h[i]/(h[i]+h[i+1]))*yn;
	ELSE
		g[i]:=(6/(h[i+1]+h[i]))*(A[i+2,2]-A[i+1,2]);
	END_IF
END_FOR



FOR i:=2 TO (n-2) DO
	u[i]:=h[i]/(h[i]+h[i+1]);
	n_[i-1]:=h[i]/(h[i-1]+h[i]);
	D[i-1,i]:=n_[i-1];
	D[i,i-1]:=u[i];
END_FOR


solve(M:=D,G:=g,X=>M_);
M[1]:=0;M[5]:=0;M[2]:=M_[1];M[3]:=M_[2];M[4]:=M_[3];
