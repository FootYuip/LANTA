# kind=functionBlock
# name=Az_Ctrl_FB
# objectId=0cb0be60-638d-4066-bce3-f581bbf54550

VAR_INPUT
    sLocal : AxisCrtlStruct;
    sCtrl : AxisCrtlStruct;
    sStatus : AxisStatus;
    sVisible : AxisVisible;
    rRealPos : REAL;
    rPosLimK : REAL;
END_VAR

VAR_OUTPUT
    logicError : INT;
END_VAR

VAR
    VelPID_1 : PID;
    VelPID_2 : PID;
    VelPID_3 : PID;
    vpid1 : vpid;
    TON_0 : TON;
    bTick : BOOL;
    tr : BOOL;
    icount1 : INT;
    icount2 : INT;
    icount3 : INT;
    bfin1 : BOOL;
    bfin2 : BOOL;
    istate : INT;
    icount : INT;
END_VAR

// --- implementation (ST) ---
(*==============================================================================
  FB: Az_Ctrl_FB（方位轴 Azimuth 控制功能块）
  与 El_Ctrl_FB 结构完全相同，差异点:
    - 操作的 GVL 变量为 Az 系列（diAzTargetVel / bAzCtrlBitX / eAzErrorcode 等）
    - positionMode 加速度为 12 deg/s²（El 为 9 deg/s²）
    - positionMode 的 Kp=0.6（El 为 0.7），本控 programBootMode Kp=0.6
    - programBootMode 参数来自 PersistentVars.rAzXxx

  速度换算公式（通用）:
    diAzTargetVel [inc/s] = deg/s × 524288 / 360
==============================================================================*)

// =============================================================================
// 段 A: 轴停止或下电时清快速停止位 bit8
// =============================================================================
IF sStatus.Disabled OR(NOT sStatus.Moving)THEN
	bAzCtrlBit8 := FALSE;
END_IF

// =============================================================================
// 段 B: 全局错误复位
// =============================================================================
IF GVL.bError_Reset THEN
	sCtrl.iMode:=6;
	sLocal.iMode:=6;
	GVL.Error_code:=0;
END_IF
IF GVL.Error_code =0 THEN
	GVL.bError_Reset:=FALSE;
END_IF

// =============================================================================
// 段 C: CANopen 心跳超时（0x8130）自动恢复序列
// =============================================================================
IF GVL.eAzErrorcode = 16#8130 AND NOT tr THEN
	GVL.bAzCtrlBit7 := TRUE;
	tr:=TRUE;
	istate:=0;
	icount:=0;
	// 写入下电控制字
	GVL.bAzCtrlBit3 := FALSE;
	GVL.bAzCtrlBit0 := FALSE;
	GVL.bAzCtrlBit1 := FALSE;
	GVL.bAzCtrlBit2 := FALSE;
END_IF
IF tr THEN
	icount1 := icount1 + 1;
END_IF
IF icount1 > 101 THEN
	icount1 := 0;
	tr := FALSE;
	GVL.bAzCtrlBit7 := FALSE;
	bfin1 := TRUE;
END_IF
IF bfin1 THEN
	icount2 := icount2 + 1;
END_IF
IF icount2 > 101 THEN
	bfin1 := FALSE;
	icount2 := 0;
END_IF

// =============================================================================
// 段 D: 主控制逻辑
// =============================================================================
IF GVL.Error_code=0 AND GVL.eAzErrorcode = 0 AND NOT GVL.bAzCtrlBit7  AND NOT tr AND NOT bfin1 AND NOT bfin2 THEN

	// =========== 远控 ===========
	IF GVL.bLocalCtrl=FALSE THEN //远控
		IF sCtrl.iMode <> 1 THEN
			istate:=0;
			icount:=0;
		END_IF
		CASE sCtrl.iMode OF

			MODES.powerOff: // 下电
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN // 运动中，先停止运动
						GVL.diAzTargetVel := 0;
						bAzCtrlBit8 := TRUE;
					ELSE
						// 写入下电控制字
						GVL.bAzCtrlBit3 := FALSE;
						GVL.bAzCtrlBit0 := FALSE;
						GVL.bAzCtrlBit1 := FALSE;
						GVL.bAzCtrlBit2 := FALSE;
					END_IF
				END_IF
				// 下电状态下持续调用 vpid1（保持内部状态稳定）
				vpid1(rKpbase:=1, rKibase:=0.01, rKpMax:=1, rKiMax:=0.01, rVelThreshold:=10,rTarget:=sCtrl.rPIDCmdPos,rFeedback:=rRealPos,rPosLimPositive:=sCtrl.rSoft_limit_Postive,
							rPosLimNegitive:=sCtrl.rSoft_limit_Negative,
								bPIDEnable:=sCtrl.bPIDEnable,rVelLim:=sCtrl.rVel_Limit,rKpScale:=0.02,rKiScale:=0.005,rElPos:=GVL.rFctPosReal_1-90,
								iThrehold:=sCtrl.rPIDThre,rPIDOutDir=>sCtrl.rPIDCmdVel);

			MODES.powerOn:  // 上电（DS402 三步序列）
				IF sStatus.Disabled THEN
					GVL.diAzTargetVel := 0;
					IF GVL.siAzMode <> 3 THEN // 上电时模式默认为3，速度模式
						GVL.siAzMode := 3;
					END_IF
					IF siAzMode = 3 THEN
						CASE istate OF
							0:
								GVL.bAzCtrlBit1 := TRUE;
								GVL.bAzCtrlBit2 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=1;
								END_IF
							1:
								GVL.bAzCtrlBit0 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=2;
								END_IF
							2:
								GVL.bAzCtrlBit3 := TRUE;
						END_CASE
					END_IF
				ELSE
					IF sStatus.Moving THEN
						GVL.diAzTargetVel := 0;
						bAzCtrlBit8 := TRUE;
					ELSE
						bAzCtrlBit8 := FALSE;
					END_IF
				END_IF

			MODES.positionMode: // 远控指向模式（加速度 12 deg/s²，Kp=0.6）
				IF NOT sStatus.Disabled THEN
					IF sCtrl.bStartAutoTrack THEN
						//自跟踪模式下，由误差电压直接驱动
						VelPID_1(rTarget:=sCtrl.rChanU + rRealPos,
								rFeedback:=rRealPos,
								rPosLimPositive:=sCtrl.rSoft_limit_Postive,
								rPosLimNegative:=sCtrl.rSoft_limit_Negative,
								rVelLim:=sCtrl.rVel_Limit,
								rKp:=1,//自跟踪PI系数3,0.02
								rKi:=0,
								rKd:=0,
								bPIDEnable:=sCtrl.bPIDEnable,
								iThreshold:= -1,
								rPIDOutput=>sCtrl.rPIDCmdVel);							
					ELSE
						//PID计算速度指令
						VelPID_1(rTarget:=sCtrl.rPosCmd,
							rFeedback:=rRealPos,
							rVelLim:=sCtrl.Pos_Vel,
							rPosLimPositive:=sCtrl.rSoft_limit_Postive,
							rPosLimNegative:=sCtrl.rSoft_limit_Negative,
							rKp:=0.6,//程引PI系数低带：1,0.002  高带：3,0.02
							rKi:=0,
							rKd:=0,
							bPIDEnable:=sCtrl.bPIDEnable,
							iThreshold:=sCtrl.rPIDThre,
							rPIDOutput=>sCtrl.rPIDCmdVel);
					END_IF
					GVL.diAzTargetVel := REAL_TO_DINT(sCtrl.rPIDCmdVel * 524288 / 360);
					GVL.diAzTargetAcc := REAL_TO_UDINT(12 * 524288 / 360);   //加速度 12 deg/s²
					GVL.diAzTargetDec := REAL_TO_UDINT(12 * 524288 / 360);   //减速度
				END_IF

			MODES.velocityMode://手控速度模式-远控模式
				IF NOT sStatus.Disabled THEN
					IF ABS(sCtrl.rVelCmd) <= sCtrl.rVel_Limit THEN
						GVL.diAzTargetVel := REAL_TO_DINT(sCtrl.rVelCmd * 524288 / 360);
					ELSIF sCtrl.rVelCmd > sCtrl.rVel_Limit THEN
						GVL.diAzTargetVel := REAL_TO_DINT(sCtrl.rVel_Limit * 524288 / 360);
					ELSE
						GVL.diAzTargetVel := REAL_TO_DINT(-sCtrl.rVel_Limit * 524288 / 360);
					END_IF
					GVL.diAzTargetAcc := REAL_TO_UDINT(sCtrl.rAcceleration * 524288 / 360);   //加速度
					GVL.diAzTargetDec := REAL_TO_UDINT(sCtrl.rDeceleration * 524288 / 360);   //减速度
				END_IF

			MODES.programBootMode://PID速度模式-正常模式 程引（变增益 vpid1）
				IF NOT sStatus.Disabled THEN
					IF sCtrl.bStartAutoTrack THEN
						// 程引+自跟踪
						vpid1(rKpbase:=Persistentvars.rAzKpBase, rKibase:=Persistentvars.rAzKiBase, rKpMax:=Persistentvars.rAzKpMax, rKiMax:=Persistentvars.rAzKiMax, rVelThreshold:=Persistentvars.rAzThre,rTarget:=sCtrl.rPIDCmdPos,rFeedback:=rRealPos,rPosLimPositive:=sCtrl.rSoft_limit_Postive,
							rPosLimNegitive:=sCtrl.rSoft_limit_Negative,
								bPIDEnable:=sCtrl.bPIDEnable,rVelLim:=sCtrl.rVel_Limit,rKpScale:=Persistentvars.rAzKpscale,rKiScale:=Persistentvars.rAzKiscale,rElPos:=GVL.rFctPosReal_1-90,
								iThrehold:=sCtrl.rPIDThre,rPIDOutDir=>sCtrl.rPIDCmdVel);							
					ELSE
						// 纯程引
						vpid1(rKpbase:=Persistentvars.rAzKpBase, rKibase:=Persistentvars.rAzKiBase, rKpMax:=Persistentvars.rAzKpMax, rKiMax:=Persistentvars.rAzKiMax, rVelThreshold:=Persistentvars.rAzThre,rTarget:=sCtrl.rPIDCmdPos,rFeedback:=rRealPos,rPosLimPositive:=sCtrl.rSoft_limit_Postive,
							rPosLimNegitive:=sCtrl.rSoft_limit_Negative,
								bPIDEnable:=sCtrl.bPIDEnable,rVelLim:=sCtrl.rVel_Limit,rKpScale:=Persistentvars.rAzKpscale,rKiScale:=Persistentvars.rAzKiscale,rElPos:=GVL.rFctPosReal_1-90,
								iThrehold:=sCtrl.rPIDThre,rPIDOutDir=>sCtrl.rPIDCmdVel);
						
					END_IF
					GVL.diAzTargetVel := REAL_TO_DINT(sCtrl.rPIDCmdVel * 524288 / 360);
					GVL.diAzTargetAcc := REAL_TO_UDINT(sCtrl.rAcceleration * 524288 / 360);   //加速度
					GVL.diAzTargetDec := REAL_TO_UDINT(sCtrl.rDeceleration * 524288 / 360);   //减速度
				END_IF

			MODES.reset://重置
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN
						GVL.diAzTargetVel := 0;
						bAzCtrlBit8 := TRUE;
					ELSE
						bAzCtrlBit8 := FALSE;
					END_IF
				END_IF

			MODES.halt://待机
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN
						GVL.diAzTargetVel := 0;
						bAzCtrlBit8 := TRUE;
					ELSE
						bAzCtrlBit8 := FALSE;
					END_IF
				END_IF
			
		END_CASE

	// =========== 本控 ===========
	ELSE // 本控
		IF sLocal.iMode <> 1 THEN
			istate:=0;
			icount:=0;
		END_IF
		CASE sLocal.iMode OF

			MODES.powerOff: // 下电
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN // 运动中，先停止运动
						GVL.diAzTargetVel := 0;
						bAzCtrlBit8 := TRUE;
					ELSE
						// 写入下电控制字
						GVL.bAzCtrlBit3 := FALSE;
						GVL.bAzCtrlBit0 := FALSE;
						GVL.bAzCtrlBit1 := FALSE;
						GVL.bAzCtrlBit2 := FALSE;
					END_IF
				END_IF

			MODES.powerOn:  // 上电
				IF sStatus.Disabled THEN
					GVL.diAzTargetVel := 0;
					IF GVL.siAzMode <> 3 THEN // 上电时模式默认为3，速度模式
						GVL.siAzMode := 3;
					END_IF
					IF siAzMode = 3 THEN
						CASE istate OF
							0:
								GVL.bAzCtrlBit1 := TRUE;
								GVL.bAzCtrlBit2 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=1;
								END_IF
							1:
								GVL.bAzCtrlBit0 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=2;
								END_IF
							2:
								GVL.bAzCtrlBit3 := TRUE;
						END_CASE
					END_IF
				ELSE
					IF sStatus.Moving THEN
						GVL.diAzTargetVel := 0;
						bAzCtrlBit8 := TRUE;
					ELSE
						bAzCtrlBit8 := FALSE;
					END_IF
				END_IF

			MODES.velocityMode://手控速度模式-本控模式
				IF NOT sStatus.Disabled THEN
					IF ABS(sLocal.rVelCmd) <= sLocal.rVel_Limit THEN
						GVL.diAzTargetVel := REAL_TO_DINT(sLocal.rVelCmd * 524288 / 360);
					ELSIF sLocal.rVelCmd > sLocal.rVel_Limit THEN
						GVL.diAzTargetVel := REAL_TO_DINT(sLocal.rVel_Limit * 524288 / 360);
					ELSE
						GVL.diAzTargetVel := REAL_TO_DINT(-sLocal.rVel_Limit * 524288 / 360);
					END_IF
					GVL.diAzTargetAcc := REAL_TO_UDINT(sLocal.rAcceleration * 524288 / 360);   //加速度
					GVL.diAzTargetDec := REAL_TO_UDINT(sLocal.rDeceleration * 524288 / 360);   //减速度
				END_IF

			MODES.programBootMode://本控位置模式（普通 PID，Kp=0.6）
				IF NOT sStatus.Disabled THEN
					IF NOT sLocal.bPIDEnable THEN
						GVL.diAzTargetVel := 0;							
					ELSE
						IF NOT slocal.bStartAutoTrack THEN
							//PID计算速度指令
							 VelPID_2(rTarget:=sLocal.rPosCmd,
									 rFeedback:=rRealPos,
									 rPosLimPositive:=sLocal.rSoft_limit_Postive,//软限位
									 rPosLimNegative:=sLocal.rSoft_limit_Negative,
									 rKp:=0.6,
									 rKi:=0,
									 rKd:=0,
									 iThreshold:=sLocal.rPIDThre,
									 bPIDEnable:=sLocal.bPIDEnable,
									 rVelLim:=sLocal.Pos_Vel,
									 rPIDOutput=>sLocal.rPIDCmdVel);//普通PID
						ELSE
							//PID计算速度指令
							 VelPID_2(rTarget:=sLocal.rChanU + rRealPos,
								rFeedback:=rRealPos,
								rPosLimPositive:=sLocal.rSoft_limit_Postive,
								rPosLimNegative:=sLocal.rSoft_limit_Negative,
								rVelLim:=sLocal.rVel_Limit,
								rKp:=1,//自跟踪PI系数3,0.02
								rKi:=0,
								rKd:=0,
								bPIDEnable:=sLocal.bPIDEnable,
								iThreshold:= -1,
								rPIDOutput=>sLocal.rPIDCmdVel);//普通PID
						END_IF
						
					END_IF
					GVL.diAzTargetVel := REAL_TO_DINT(sLocal.rPIDCmdVel * 524288 / 360);
					GVL.diAzTargetAcc := REAL_TO_UDINT(12 * 524288 / 360);   //加速度
					GVL.diAzTargetDec := REAL_TO_UDINT(12 * 524288 / 360);   //减速度
				END_IF

			MODES.reset://重置
			// 是否需要
			MODES.halt://待机
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN
						GVL.diAzTargetVel := 0;
						bAzCtrlBit8 := TRUE;
					ELSE
						bAzCtrlBit8 := FALSE;
					END_IF
				END_IF
			
		END_CASE
	END_IF

// =============================================================================
// 段 E: 全局错误时强制停轴，超时后强制下电
// =============================================================================
ELSIF GVL.Error_code<>0 AND GVL.eAzErrorcode = 0 AND NOT GVL.bAzCtrlBit7  AND NOT tr AND NOT bfin1 AND NOT bfin2 THEN
	// 出现错误，轴停止运动
	GVL.diAzTargetVel := 0;
	bAzCtrlBit8 := TRUE;
	TON_0(IN:=bAzCtrlBit8,PT:=T#1S,Q=>,ET=>);//判断stop0执行是否超时,若超时强制下电（手动复位）
	IF TON_0.Q THEN
		bTick:=TRUE;
	END_IF
	IF bTick AND sStatus.Moving THEN // 超时强制下电
		// 写入下电控制字
		GVL.bAzCtrlBit3 := FALSE;
		GVL.bAzCtrlBit0 := FALSE;
		GVL.bAzCtrlBit1 := FALSE;
		GVL.bAzCtrlBit2 := FALSE;
	END_IF
END_IF
