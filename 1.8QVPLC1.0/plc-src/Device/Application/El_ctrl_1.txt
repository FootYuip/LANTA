# kind=program
# name=El_ctrl_1
# objectId=606036f3-9e29-4a9f-a89f-c2eaefee70cf

VAR
    Axis1_Ctrl : El_Ctrl_FB;
    bRemote2LocalStop : BOOL;
    bLocal2RemoteStop : BOOL;
    Error : INT;
    Axis1Limit_pos : REAL;
    Axis1Limit_Neg : REAL;
    bRTriger : Standard.R_TRIG;
    bFTriger : Standard.F_TRIG;
    Stop1_Count : INT;
    bStop_1 : BOOL;
    rPosModeStepBias : LREAL;
    rPosModeStep : LREAL;
    rStepBiasMax : LREAL;
    rStepBias : LREAL;
    rStep : LREAL;
    rAuToStepBias : LREAL;
    rAuToStepBiasTemp : LREAL;
    rAutoStep : LREAL;
    TbAutoTrigger : Standard.F_TRIG;
    RbAutoTrigger : Standard.R_TRIG;
    rAutoTrackingBiasMax : LREAL;
    rLocalInitPosModeSpeed : REAL;
    rRemoteInitPosModeSpeed : REAL;
    iCountOffAuto : INT;
    iCasePBAutoTrack : INT;
    rKAuToStepBias : LREAL;
    agcmaxsearch_tb : AGCMaxSearch_PB_1;
    rSearchNextStepTarget_tb : REAL;
    agcmaxsearch_pb : AGCMaxSearch_PB_1;
    rElthreshold : REAL;
    toN1 : Standard.Ton;
    ttime1 : TIME;
    bstartagcmaxsearch : BOOL;
    rSearchNextStepTarget : REAL;
    rMaxPosTest : REAL;
END_VAR

// --- implementation (ST) ---
(*==============================================================================
  PRG: El_ctrl_1（俯仰轴 El 顶层程序）
  任务: Canopen 10ms
  作用:
    1. 读 CANopen 反馈 → 换算实际速度/位置 (rFctVelReal_1 / rFctPosReal_1)
    2. 组装 Axis1Status（故障、使能、运动中等）
    3. 远/本控切换保护、自跟踪拉偏、AGC 峰值搜索
    4. 合成 PID 目标位置 rPIDCmdPos → 调用 El_Ctrl_FB
  数据流:
    Modbus/上位机 → Axis1Remote（在 Modbus_Slave 等处赋值）
    RealTimePosCal.rInterElPos + 各类偏置 → rPIDCmdPos → El_Ctrl_FB → GVL.diEl*
==============================================================================*)




(* 远控参量曾在此从 Modbus_Slave 直赋，现改由 Modbus_Slave 程序统一写入 Axis1Remote *)

// =============================================================================
// 段 1: 驱动状态字 → Axis1Status（供 El_Ctrl_FB 与 HMI 使用）
// DS402 状态: bit0/1/2 同时为 1 表示 Operation Enabled（伺服已使能）
// =============================================================================
Axis1Status.Error := GVL.bElStsBit3;
Axis1Status.ErrorID := GVL.eElErrorcode;
Axis1Status.bServo_Enable := GVL.bElStsBit0 AND GVL.bElStsBit1 AND GVL.bElStsBit2;
Axis1Status.bBrake := NOT (GVL.bElStsBit0 AND GVL.bElStsBit1 AND GVL.bElStsBit2);
Axis1Status.Disabled := NOT (GVL.bElStsBit0 AND GVL.bElStsBit1 AND GVL.bElStsBit2);
// 软限位写入命令结构体（与下面硬限位 90~180 一致）
Axis1Remote.rSoft_limit_Negative:=90;Axis1Remote.rSoft_limit_Postive:=180;
Axis1Local.rSoft_limit_Negative:=90;Axis1Local.rSoft_limit_Postive:=180;

// =============================================================================
// 段 2: CANopen 反馈 → 实际速度/位置（deg, deg/s）
// 公式: 物理量 = diActXxx / 524288 × 360；俯仰另加 Modbus 标定偏移 lrElOffset
// =============================================================================
rFctVelReal_1:=DINT_TO_REAL(diElActVel)/524288.0*360.0;
rFctPosReal_1:=DINT_TO_REAL(diElActPos)/524288.0*360.0 + modbus_slave.lrElOffset;

// =============================================================================
// 段 3: 指向命令硬限位（远控/本控 rPosCmd 钳位到 90°~180°）
// =============================================================================
Axis1Limit_pos:=180;
Axis1Limit_Neg:=90;
IF Axis1Remote.rPosCmd>Axis1Limit_pos THEN
	Axis1Remote.rPosCmd:=180;
ELSIF Axis1Remote.rPosCmd<Axis1Limit_Neg THEN
	 Axis1Remote.rPosCmd:=90;  
END_IF

IF Axis1Local.rPosCmd>Axis1Limit_pos THEN
	Axis1Local.rPosCmd:=180;
ELSIF Axis1Local.rPosCmd<Axis1Limit_Neg THEN
	 Axis1Local.rPosCmd:=90;  
END_IF

// =============================================================================
// 段 4: 停稳判定 → Axis1Status.Moving
// |速度|<0.035°/s 连续 100 拍(约1s) 认为停稳；bStop_1=TRUE → Moving=FALSE
// =============================================================================
IF  ABS(rFctVelReal_1)<0.035 THEN
	Stop1_Count:=Stop1_Count+1;
	IF Stop1_Count>100 THEN
		Stop1_Count:=101;
		bStop_1:=TRUE;;
	END_IF
ELSE
	Stop1_Count:=0;
	bStop_1:=FALSE;
END_IF
Axis1Status.Moving := NOT bStop_1;

// =============================================================================
// 段 5: PID 使能 — 仅指向(2)与程引(4)模式需要位置环
// =============================================================================
IF Axis1Remote.iMode=2 OR Axis1Remote.iMode=4 THEN
	Axis1Remote.bPIDEnable:=TRUE;
ELSE
	Axis1Remote.bPIDEnable:=FALSE;
END_IF

// =============================================================================
// 段 6: 远控 ↔ 本控 切换保护（GVL.bLocalCtrl）
// 切换瞬间若伺服仍使能 → 置切换标志 → 强制 mode=7(halt) 减速
// 停稳且下电完成后 → mode=0，清除标志
// =============================================================================
bRTriger(CLK:=GVL.bLocalCtrl);
bFTriger(CLK:=GVL.bLocalCtrl);

// 远控 → 本控（bLocalCtrl 上升沿）
IF Axis1Status.bServo_Enable AND (bRTriger.Q ) THEN//如果速度不为0且本控发生切换，则减速停止标识为True
	bRemote2LocalStop:=TRUE;
END_IF
IF bRemote2LocalStop  THEN //若减速停止标识符为True 
	
	IF  bStop_1=TRUE THEN//且电机速度不为0 开始减速 减速完成后下使能，下使能完成后复位标识符复位
		Axis1Local.iMode:=0;
		Axis1Remote.iMode:=0;
		IF Axis1Status.bServo_Enable=FALSE THEN
			bRemote2LocalStop:=FALSE;
		END_IF
	ELSE
		Axis1Local.iMode:=7;
		Axis1Remote.iMode:=7;
	END_IF
END_IF

// 本控 → 远控（bLocalCtrl 下降沿）
IF Axis1Status.bServo_Enable AND (bFTriger.Q ) THEN//如果速度不为0且本控发生切换，则减速停止标识为True
	bLocal2RemoteStop:=TRUE;
END_IF
IF bLocal2RemoteStop  THEN //若减速停止标识符为True 
	
	IF  bStop_1=TRUE THEN//且电机速度不为0 开始减速 减速完成后下使能，下使能完成后复位标识符复位
		Axis1Local.iMode:=0;
		Axis1Remote.iMode:=0;
		IF Axis1Status.bServo_Enable=FALSE THEN
			bLocal2RemoteStop:=FALSE;
		END_IF
	ELSE
		Axis1Remote.iMode:=7;
		Axis1Local.iMode:=7;
	END_IF
END_IF

// =============================================================================
// 段 7: AGC 峰值搜索（与方位 Az_ctrl_1 协同，bstartSearchEl 由方位触发）
// mode2: 插值目标 = rPosCmd；mode4: 插值目标 = RealTimePosCal.rInterElPos
// 仅当俯仰高于 rElthreshold（相对水平）且在程引下才在 mode4 启用搜索
// =============================================================================
IF Axis1Remote.iMode=2 THEN
			IF GVL.Chan1AGC>=(GVL.Chanthreshold-0.1) THEN
					agcmaxsearch_pb(bstart:=Az_ctrl_1.agcmaxsearch_tb.bSucess,rAGCNow:=GVL.Chan1AGC,btrig:=Az_ctrl_1.bstartSearchEl,
						rCurrentPos:=rFctPosReal_1,rInterPos:=Axis1Remote.rPosCmd,
				rStepBiasNow:=rStepBias,bAutoTracking:=GVL.bStartTrack,bStop:=FALSE,
				rStepBias=>rSearchNextStepTarget_tb,rAGCMaxBias=>rMaxPosTest,bSucess=>);
			ELSE
				bstartagcmaxsearch:=FALSE;
				agcmaxsearch_pb(bstart:=Az_ctrl_1.agcmaxsearch_tb.bSucess,rAGCNow:=GVL.Chan1AGC,btrig:=Az_ctrl_1.bstartSearchEl,
				rCurrentPos:=rFctPosReal_1,rInterPos:=Axis1Remote.rPosCmd,
				rStepBiasNow:=rStepBias,bAutoTracking:=GVL.bStartTrack,bStop:=FALSE,
				rStepBias=>rSearchNextStepTarget_tb,rAGCMaxBias=>rMaxPosTest,bSucess=>);
			END_IF
		ELSIF Axis1Remote.iMode=4 AND rFctPosReal_1-90>rElthreshold  THEN
			IF GVL.Chan1AGC>=(GVL.Chanthreshold-0.1) THEN
					agcmaxsearch_pb(bstart:=Az_ctrl_1.agcmaxsearch_pb.bSucess,rAGCNow:=GVL.Chan1AGC,btrig:=Az_ctrl_1.bstartSearchEl,
						rCurrentPos:=rFctPosReal_1,rInterPos:=RealTimePosCal.rInterElPos,
				rStepBiasNow:=rStepBias,bAutoTracking:=GVL.bStartTrack,bStop:=FALSE,
				rStepBias=>rSearchNextStepTarget,rAGCMaxBias=>rMaxPosTest,bSucess=>);
			ELSE
				bstartagcmaxsearch:=FALSE;
				agcmaxsearch_pb(bstart:=Az_ctrl_1.agcmaxsearch_pb.bSucess,rAGCNow:=GVL.Chan1AGC,btrig:=Az_ctrl_1.bstartSearchEl,
				rCurrentPos:=rFctPosReal_1,rInterPos:=RealTimePosCal.rInterElPos,
				rStepBiasNow:=rStepBias,bAutoTracking:=GVL.bStartTrack,bStop:=FALSE,
				rStepBias=>rSearchNextStepTarget,rAGCMaxBias=>rMaxPosTest,bSucess=>);
			END_IF
		ELSIF Axis1Remote.iMode =0  OR Axis1Remote.iMode =7 THEN
			bstartagcmaxsearch:=FALSE;
			agcmaxsearch_pb(bstart:=Az_ctrl_1.agcmaxsearch_pb.bSucess,rAGCNow:=GVL.Chan1AGC,btrig:=Az_ctrl_1.bstartSearchEl,
			rCurrentPos:=rFctPosReal_1,rInterPos:=RealTimePosCal.rInterElPos,	
			rStepBiasNow:=rStepBias,bAutoTracking:=GVL.bStartTrack,bStop:=FALSE,
			rStepBias=>rSearchNextStepTarget,rAGCMaxBias=>rMaxPosTest,bSucess=>);
		END_IF

// =============================================================================
// 段 8: 自跟踪拉偏（远控）
// mode4+程引: 用 AutoTrackingCal.rChanUe 累积 rAuToStepBias（限幅）
// mode2+指向: 直接用 rChanU 作 PID 目标增量（在 El_Ctrl_FB 内 rChanU+位置）
// 关跟踪: rAuToStepBias 渐零，rStepBias 渐近 Modbus_Slave.rBias_1
// =============================================================================
IF GVL.bStartTrack AND Axis1Remote.iMode=4 THEN
	Axis1Remote.bStartAutoTrack := TRUE;
	IF rAuToStepBias > rAutoTrackingBiasMax THEN
		rAuToStepBias := rAutoTrackingBiasMax;
	ELSIF rAuToStepBias < -rAutoTrackingBiasMax THEN
		rAuToStepBias := -rAutoTrackingBiasMax;
	ELSE
		rAuToStepBiasTemp:=0.02*AutoTrackingCal.rDirSenFac*AutoTrackingCal.rChanUe;
		rAuToStepBias:=rAuToStepBias+AutoTrackingCal.rDirSenFac*rKAuToStepBias*AutoTrackingCal.rChanUe/1000.0;
	END_IF
ELSIF NOT GVL.bStartTrack AND Axis1Remote.iMode=4 THEN
	IF 	rAuToStepBias>0 THEN//关闭自跟踪后自跟踪拉偏量逐渐归0
		rAuToStepBias:=rAuToStepBias-rAutoStep;
	ELSIF rAuToStepBias<0 THEN
		rAuToStepBias:=rAuToStepBias+rAutoStep;
	ELSE
		rAuToStepBias:=0;
	END_IF
	
	IF Modbus_Slave.rBias_1>rStepBias THEN
		rStepBias:=rStepBias+rStep;
		IF rStepBias>=Modbus_Slave.rBias_1 THEN
			rStepBias:=Modbus_Slave.rBias_1;
		END_IF
	ELSIF Modbus_Slave.rBias_1<rStepBias THEN
		rStepBias:=rStepBias-rStep;
		IF rStepBias<=Modbus_Slave.rBias_1 THEN
			rStepBias:=Modbus_Slave.rBias_1;
		END_IF
	END_IF
	rAuToStepBiasTemp:=0;
ELSIF GVL.bStartTrack AND Axis1Remote.iMode=2 THEN
	Axis1Remote.bStartAutoTrack := TRUE;
	Axis1Remote.rChanU := AutoTrackingCal.rChanUe;
ELSE
	Axis1Remote.bStartAutoTrack := FALSE;	
	rAuToStepBias:=0;
	rStepBias:=0;
	rAuToStepBiasTemp:=0;
END_IF

// 本控自跟踪（仅程引 mode4）
IF GVL.bStartLocalAutoTrack AND Axis1Local.iMode = 4 THEN
	Axis1Local.bStartAutoTrack := TRUE;
	Axis1Local.rChanU := AutoTrackingCal.rChanUeLocal;
ELSE
	Axis1Local.bStartAutoTrack := FALSE;
END_IF

// =============================================================================
// 段 9: 合成 PID 目标位置并限位 → 调用 El_Ctrl_FB
// rPIDCmdPos = 程引插值 + 手动偏置 + 自跟踪偏置 + AGC搜索偏置 + 安全检查偏置
// =============================================================================
Axis1Remote.rPIDCmdPos:=RealTimePosCal.rInterElPos+rStepBias+rAuToStepBias+rSearchNextStepTarget+SECURITYCHECK.el_out+rAuToStepBiasTemp;
IF Axis1Remote.rPIDCmdPos<90 THEN
	Axis1Remote.rPIDCmdPos:=90;
ELSIF Axis1Remote.rPIDCmdPos>180 THEN
	Axis1Remote.rPIDCmdPos:=180;
END_IF

Axis1_Ctrl(sLocal:=Axis1Local, sCtrl:=Axis1Remote, sStatus:=Axis1Status, sVisible:=Axis1Visible, rRealPos:=rFctPosReal_1, rPosLimK:=0.34);
