(*==============================================================================
  PRG: Modbus_Slave（Modbus TCP 从站网关）
  任务: MainTask 50ms
  协议: Modbus TCP 端口 502，GCAN FB_ModbusTCP_Slave_*

  内存映射（与 GVL AT %MW 一致）:
    modbusData      AT %MW0   — 保持寄存器 HR，上位机读写（下发命令+多项式系数）
    modbusInputRegs AT %MW660 — 输入寄存器 IR，PLC 写入、上位机只读（状态上报）

  数据流:
    上位机写 HR → 本程序解析 → iMode_x/rPosCmd_x/aUTC_TIME 等
                              → SecurityCheck 写 AxisXRemote
    PLC 状态 → 打包 IR → 上位机读

  扩展自定义寄存器: 在 HR 区 modbusData[n] 增加解析/在 IR 区赋值即可
==============================================================================*)
// =============================================================================
// 段 A: Modbus TCP 从站初始化（端口 502，仅首次）
// =============================================================================
IF isInit=FALSE THEN							
	
	modtcpslaveinit(en_in:=TRUE,					//Modbus TCP Slave初始化功能块使能
	xenable:=TRUE,									//使能初始化功能
	iport:=502,										//开放端口号502
	en_out=>eout,									//Modbus TCP Slave初始化功能块执行完成
	isocket=>iSocket,								//创建的socket存入变量
	uierror=>uiErr);								//Modbus TCP Slave初始化功能块执行错误信息
	
	IF isocket>=0 AND uiErr=0 THEN					//成功创建socket且无错误，逻辑标志置位，下周期执行ELSE
		isInit:=TRUE;	
		
		enCtr:=FALSE;
		ELSE                                        //等待连接成功
		modtcpslaveinit(en_in:=0,xenable:=0);	
	END_IF
	
 
ELSE

// =============================================================================
// 段 B: 从站周期服务 — 映射 HR/IR 缓冲区，看门狗 2000ms
// =============================================================================
	modtcpslavectrl(en_in:=enCtr,					//Modbus TCP Slave执行功能块使能
	xenable:=TRUE,									//Modbus TCP Slave执行功能使能

	uierror=>uierr);								//Modbus TCP Slave执行功能块执行错误信息
	enctr:=TRUE;
	IF uierr <>0 THEN                               //如果从站报错，则重新连接从站

	END_IF
	//iMode:  0 下电  1 上电  2 速度模式  3 停止  4 位置模式  5 继续运行
	//方位轴指令
(*
	Az_ctrl.iMode:=WORD_TO_INT(modbusData[0]);
	AZ_ctrl.rTarVel:=WORD_TO_REAL(modbusData[1]);
	AZ_ctrl.rTarPos:=WORD_TO_REAL(modbusData[2])*-1.0;

	El_ctrl.iMode:=WORD_TO_INT(modbusData[10]);
	El_ctrl.rTarVel:=WORD_TO_REAL(modbusData[11]);
	El_ctrl.rTarPos:=WORD_TO_REAL(modbusData[12])+90.0;
	*)

	// HR[0..3] UTC  HR[7] 控制字  HR[8..18]Az  HR[19..29]El  HR[30..39]Ti
	// HR[43] AGC门限  HR[45..48]轨道时间  HR[51..54]拉偏  HR[57..58]俯仰零位
	// =============================================================================
	UTC_Union.Apart_word[1]:=modbusData[3];
	UTC_Union.Apart_word[2]:=modbusData[2];
	UTC_Union.Apart_word[3]:=modbusData[1];
	UTC_Union.Apart_word[4]:=modbusData[0];
	UTC_New:=UTC_Union.Apart_ulint;
	DeltaTimeBetPC := UTC_New - TO_ULINT(GVL.dRealTime);
	// HR[7] 控制字位定义（WORD_AS_BIT）
	CtrlBool(W:=modbusData[7],
			B14=>bCommReset,

			B11=>Axis3Remote.bPIDEnable,

			);
	//方位轴指令
	iMode_2:=WORD_TO_INT(modbusData[8]);

	Pos2_Union.Apart_word[1]:=modbusData[10];

	rP_2:=INT_TO_LREAL(WORD_TO_INT(modbusData[12]))/1000;
	rI_2:=INT_TO_LREAL(WORD_TO_INT(modbusData[13]))/1000;
	rD_2:=INT_TO_LREAL(WORD_TO_INT(modbusData[14]))/1000;
	rThreshold_2:=WORD_TO_INT(modbusData[15])/1000;
	rAcceleration_2:=INT_TO_LREAL(WORD_TO_INT(modbusData[16]))/1000;
	rDeceleration_2:=INT_TO_LREAL(WORD_TO_INT(modbusData[17]))/1000;
	rVelLim_2:=INT_TO_LREAL(WORD_TO_INT(modbusData[18]))/1000;
	
	//俯仰轴指令
	iMode_1:=WORD_TO_INT(modbusData[19]);
	Pos1_Union.Apart_word[2]:=modbusData[20];
	Pos1_Union.Apart_word[1]:=modbusData[21];
	rPosCmd_1:=DINT_TO_LREAL( DWORD_TO_DINT( Pos1_Union.Apart_dword[1]))/100000+90;
	rVelcmd_1:=DINT_TO_LREAL( WORD_TO_INT(modbusData[22]))/1000;
	rP_1:=INT_TO_LREAL(WORD_TO_INT(modbusData[23]))/1000;
	rI_1:=INT_TO_LREAL(WORD_TO_INT(modbusData[24]))/1000;
	rD_1:=INT_TO_LREAL(WORD_TO_INT(modbusData[25]))/1000;
	rThreshold_1:=WORD_TO_INT(modbusData[26])/1000;
	rAcceleration_1:=INT_TO_LREAL(WORD_TO_INT(modbusData[27]))/1000;
	rDeceleration_1:=INT_TO_LREAL(WORD_TO_INT(modbusData[28]))/1000;
	rVelLim_1:=INT_TO_LREAL(WORD_TO_INT(modbusData[29]))/1000;
	
	//倾斜轴指令
	iMode_3:=WORD_TO_INT(modbusData[30]);
	Pos3_Union.Apart_word[2]:=modbusData[31];
	Pos3_Union.Apart_word[1]:=modbusData[32];
	rPosCmd_3:=DINT_TO_LREAL( DWORD_TO_DINT( Pos3_Union.Apart_dword[1]))/100000*-1;
	rVelcmd_3:=DINT_TO_LREAL( WORD_TO_INT(modbusData[33]))/1000*-1;
	rP_3:=INT_TO_LREAL(WORD_TO_INT(modbusData[34]))/1000;
	rI_3:=INT_TO_LREAL(WORD_TO_INT(modbusData[35]))/1000;
	rD_3:=INT_TO_LREAL(WORD_TO_INT(modbusData[36]))/1000;
	rThreshold_3:=WORD_TO_INT(modbusData[37])/1000;
	rAcceleration_3:=INT_TO_LREAL(WORD_TO_INT(modbusData[38]))/1000;
	rDeceleration_3:=INT_TO_LREAL(WORD_TO_INT(modbusData[39]))/1000;
	
	
	//AGC锁定门限
	AGC_Value:=(WORD_TO_UDINT(modbusData[43])/100.0);
	GVL.Chanthreshold:=AGC_Value;
	
	//轨道时间
	UTC_Orbit_Union.Apart_word[1]:=modbusData[48];
	UTC_Orbit_Union.Apart_word[2]:=modbusData[47];
	UTC_Orbit_Union.Apart_word[3]:=modbusData[46];
	UTC_Orbit_Union.Apart_word[4]:=modbusData[45];
	UTC_Orbit_New:=UTC_Orbit_Union.Apart_ulint;
	//时间拉偏
	TimeBias_Union.Apart_word[1]:=modbusData[50];
	TimeBias_Union.Apart_word[2]:=modbusData[49];
	rTimeBias:=DINT_TO_LREAL( DWORD_TO_DINT( TimeBias_Union.Apart_dword[1]))/100000.0; //单位为秒
	IF rTimeBias>0.4 THEN//时间偏置最大限制 正负0.4秒
		rTimeBias:=0.4;
	ELSIF rTimeBias<-0.4 THEN
		rTimeBias:=-0.4;
	END_IF
	rTimeBias:=rTimeBias*1000000000;//秒转纳秒
	//两轴拉偏
	Bias2_Union.Apart_word[1]:=modbusData[52];
	Bias2_Union.Apart_word[2]:=modbusData[51];
	rBias_2:=DINT_TO_LREAL( DWORD_TO_DINT( Bias2_Union.Apart_dword[1]))/100000.0*(-1);
	
	Bias1_Union.Apart_word[1]:=modbusData[54];
	Bias1_Union.Apart_word[2]:=modbusData[53];
	rBias_1:=DINT_TO_LREAL( DWORD_TO_DINT( Bias1_Union.Apart_dword[1]))/100000.0;
	
	//俯仰零值
	AcuElOffset.Apart_word[1]:=modbusData[58];
	AcuElOffset.Apart_word[2]:=modbusData[57];
	lrElOffset:=DINT_TO_LREAL( DWORD_TO_DINT( AcuElOffset.Apart_dword[1]))/100000.0;
	// ----- 程引轨道表: UTC_Orbit_New 递增时移位写入 aUTC_TIME / aPosCmd_x -----
	
	// 指向模式最大速度 HR[61..63] → GVL.AxisXRemote.Pos_Vel
	rAzPosVel:=INT_TO_LREAL(WORD_TO_INT(modbusData[61]))/1000.0;
	rElPosVel:=INT_TO_LREAL(WORD_TO_INT(modbusData[62]))/1000.0;
	rTlPosVel:=INT_TO_LREAL(WORD_TO_INT(modbusData[63]))/1000.0;
	
	IF rAzPosVel <= 16.0 THEN
		GVL.Axis2Remote.Pos_Vel := rAzPosVel;
	ELSE
		GVL.Axis2Remote.Pos_Vel := 16;
	END_IF
	
	IF rElPosVel <= 11.0 THEN
		GVL.Axis1Remote.Pos_Vel := rElPosVel;
	ELSE
		GVL.Axis1Remote.Pos_Vel := 11;
	END_IF
	
	IF rTlPosVel <= 8.0 THEN
		GVL.Axis3Remote.Pos_Vel := rTlPosVel;
	ELSE
		GVL.Axis3Remote.Pos_Vel := 8;
	END_IF
	
	FOR iCheck:=1 TO 10 DO
		IF UTC_Orbit_New>aUTC_TIME[iCheck] THEN
			bPass:=TRUE;
		ELSE
			bPass:=FALSE;
			EXIT;
		END_IF
	END_FOR
	IF bPass THEN //命令刷新，则更新序列
		FOR isave:=1 TO 9 DO
			aUTC_TIME[isave]:=aUTC_TIME[isave+1];
			aPosCmd_1[isave]:=aPosCmd_1[isave+1];
			aPosCmd_2[isave]:=aPosCmd_2[isave+1];
		END_FOR
		aUTC_TIME[10]:=UTC_Orbit_New;
		aPosCmd_1[10]:=rPosCmd_1;
		aPosCmd_2[10]:=rPosCmd_2;
	END_IF
	//若不是程引模式，则清空轨道时间数据
	IF Axis1Remote.iMode<>4 AND Axis2Remote.iMode<>4 THEN
		FOR iCheck:=1 TO 10 DO
			aUTC_TIME[iCheck]:=0;;
		END_FOR
	END_IF
	
	// =============================================================================
	// 段 D: 打包 IR 上报（modbusInputRegs，上位机只读）
	// IR[0..3]时间 IR[5]Error IR[6..9]DI/DO IR[10+]三轴状态 IR[59..65]跟踪机
	// IR[69..74]跟踪偏差 IR[75..78]理论角
	// =============================================================================
	IF icount<16#FFFF THEN
		icount:=icount+1;
	ELSE
		icount:=0;
	END_IF
	
	DI1_Union(B15:=SecurityCheck.bPrelimit_Positive_1,						//俯仰轴预限位--正 B15=0为未触发，B15=1为触发
			B14:=SecurityCheck.bFinallimit_1,								//俯仰轴终限位
			B13:=SecurityCheck.bPrelimit_Negative_1,						//俯仰轴预限位--负
			B12:=SecurityCheck.bFinallimit_1,								//俯仰位终限位
			B11:=SecurityCheck.bPrelimit_Positive_2,						//方位轴预限位--正
			B10:=SecurityCheck.bFinallimit_2,								//方位轴终限位
			B09:=SecurityCheck.bPrelimit_Negative_2,						//方位轴预限位--负
			B08:=SecurityCheck.bFinallimit_2,								//方位轴终限位
			B07:=FALSE,												//倾斜预限位--正
			B06:=FALSE,												//倾斜终限位--正
			B05:=FALSE,												//倾斜预限位--负
			B04:=FALSE,												//倾斜终限位--负
			B03:=FALSE,												//方位轴过零检测
			B02:=Axis2Status.bBrake,								//方位轴抱闸状态 0为解抱闸，1为抱闸
			B01:=Axis1Status.bBrake,								//俯仰轴抱闸状态
			B00:=Axis3Status.bBrake,								//倾斜轴抱闸状态
			W=>DI1_word	);
			
	DI2_Union(B15:=FALSE,								//
			B14:=FALSE,								//
			B13:=FALSE,
			B12:=FALSE,
			B11:=FALSE,
			B10:=FALSE,
			B09:=FALSE,
			B08:=FALSE,
			B07:=FALSE,
			B06:=FALSE,
			B05:=GVL.bError_Reset,											//错误复位符
			B04:=GVL.bLocalCtrl,											//本远控标识符
			B03:=FALSE,									//退限位标识符
			B02:=Axis2Remote.bPIDEnable,											//方位轴PID使能状态
			B01:=Axis1Remote.bPIDEnable,											//俯仰轴PID使能状态
			B00:=Axis3Remote.bPIDEnable,											//倾斜轴PID使能状态
			W=>DI2_word);
	DO1_Union(B15:=FALSE,								//中心体急停指示灯
			B14:=FALSE,							//声光报警		1为触发，0为未触发
			B13:=FALSE,							//爬梯急停指示灯 B13=1为亮 B13=0为灭
			B12:=FALSE,							//机柜蜂鸣器	B12=1为触发，B12=0为未触发
			B11:=FALSE,							//限位释放指示灯 同上 
			B10:=FALSE,						//本远控指示灯	B10=1为本控，B10=0为远控
			B09:=FALSE,							//系统状态指示灯 B09=1为异常，B09=0为正常	
			B08:=Axis2Status.bBrake,								//方位轴抱闸输出 B08=1为抱闸输出 B08=1为解抱闸输出
			B07:=Axis1Status.bBrake,								//俯仰轴抱闸输出 同上
			B06:=Axis3Status.bBrake,								//倾斜轴抱闸输出 同上
			B05:=FALSE,										//方位风扇输出	B05=1为风扇运行，B05=0为风扇未运行
			B04:=FALSE,										//俯仰风扇输出	同上
			B03:=FALSE,										//倾斜风扇输出	同上
