VAR
    bPrelimit_Positive_1 : BOOL;
    bPrelimit_Negative_1 : BOOL;
    bPrelimit_Positive_2 : BOOL;
    bPrelimit_Negative_2 : BOOL;
    bPrelimit_1 : BOOL;
    bPrelimit_2 : BOOL;
    bFinallimit_1 : BOOL;
    bFinallimit_2 : BOOL;
    AzError : REAL;
    ElError : REAL;
    rELCurrent : REAL;
    rAzCurrent : REAL;
    rTlCurrent : REAL;
    rAz_realpos : REAL;
    rEL_realpos : REAL;
    fbScan : SpiralScan;
    fbScan1 : SpiralScan_1;
    fbScantest : SpiralScan;
    enabletest : BOOL;
    aztest : REAL;
    eltest : REAL;
    enable : BOOL;
    delta : REAL;
    n_steps : REAL;
    agc_sim : REAL;
    lock_sim : BOOL;
    cycle_cnt : INT;
    sim_done : BOOL;
    az_out : REAL;
    el_out : REAL;
    done_out : BOOL;
    bLockSearch : BOOL;
    rSearchThre : REAL;
    rSearchThreTemp : REAL;
    bStartTrackTemp : BOOL;
    RiseEdge : R_TRIG;
    FallEdge : F_TRIG;
    bAcuTrackInhibit : BOOL;
    icountLostLocking : INT;
    bErrorLock : BOOL;
    icountLostLocking1 : INT;
    bType_out : BOOL;
    iconutwait : INT;
    rdelta : REAL := 0.25;         // 自动计算的螺旋每格边长：波束角/4
    rStep : REAL := 12.0;          // 自动读取的螺旋每格周期数
    rSpiralFullTimeSec : REAL;     // 不提前锁定时，完整螺旋轨迹预计时间(s)
    bFinish : BOOL;
    rAzErrorReal : REAL;
    rElErrorReal : REAL;
    rAlpha : REAL;
    icountBaseLock : INT;
    bScanRun : BOOL;
    bScanDoneOnce : BOOL;
    bScanFound : BOOL;
    az_latch : REAL;
    el_latch : REAL;
    bBaseLock : BOOL;
    rTrackErrorMag : REAL;
    rTrackReleaseThreshold : REAL := 1.0; // 自跟踪稳定后释放扫描偏置的误差电压门限(V)
    rScanReleaseStep : REAL := 0.0002;    // 每10ms释放的扫描偏置(°)，须小于自跟踪可补偿速率
    iTrackStableNeed : INT := 20;         // 连续稳定拍数，20拍=200ms
    iTrackStableCount : INT;
    bEnableAgcSearch : BOOL := FALSE;     // 不使用 AGC 搜索；仅保留显式恢复开关
    bAutoTrackLatch : BOOL;               // 自动转跟模式内部锁存，和ACU转跟命令完全隔离
    bTrackingModeLast : BOOL;             // 检测ACU/自动转跟模式切换
    iBaseLockLostCount : INT;             // 自动转跟后的基带连续失锁计数
    iBaseLockLostNeed : INT := 200;        // 200拍×10ms=2s，连续失锁后退出自跟踪
END_VAR

// --- implementation (ST) ---
// =============================================================================
// 段 1: 安全逻辑用位置（与显示用 rFctPosReal 可能差符号/零点）
// =============================================================================
rAz_realpos := -1 * GVL.rFctPosReal_2;
rEl_realpos := GVL.rFctPosReal_1 - 90;

// =============================================================================
// 段 2: 三轴伺服电流监测（仅使能时读 AD，%IW×%ID/1e6）
// %IW10/%ID4=俯仰, %IW20/%ID9=方位, %IW30/%ID14=倾斜
// =============================================================================
IF Axis1Status.bServo_Enable THEN
	rELCurrent:=DINT_TO_REAL((INT_TO_DINT(%IW10)* %ID4))/1000000.0;  //俯仰轴电流监测
ELSE
	rELCurrent:=0;
END_IF

IF Axis2Status.bServo_Enable THEN
	rAzCurrent:=DINT_TO_REAL((INT_TO_DINT(%IW20)* %ID9))/1000000.0;  //方位轴电流监测
ELSE
	rAzCurrent:=0;
END_IF

IF Axis3Status.bServo_Enable THEN
	rTlCurrent:=DINT_TO_REAL((INT_TO_DINT(%IW30)* %ID14))/1000000.0;  //倾斜轴电流监测
ELSE
	rTlCurrent:=0;
END_IF

// =============================================================================
// 段 3: 跟踪误差（供 SpiralScan 判断扫描方向与完成度）
// mode2 指向: 命令位置 vs 实际位置
// mode4 程引: 插值目标+偏置 vs 实际位置（含手动/自跟踪偏置）
// =============================================================================
IF Axis2Remote.iMode = 2 THEN
	AzError:=(-1) * (Axis2Remote.rPosCmd-rFctPosReal_2);
	ElError:=Axis1Remote.rPosCmd-rFctPosReal_1;	
END_IF

IF Axis2Remote.iMode = 4 THEN
	AzError:=rFctPosReal_2-RealTimePosCal.rInterazPos-(az_ctrl_1.rStepBias+az_ctrl_1.rAuToStepBias);
	ElError:=RealTimePosCal.rInterElPos-rFctPosReal_1+ (El_ctrl_1.rStepBias+EL_ctrl_1.rAuToStepBias);
END_IF
(*
rAzErrorReal := rAzErrorReal + rAlpha * (AzError - rAzErrorReal);
rElErrorReal := rElErrorReal + rAlpha * (ElError - rElErrorReal);
*)

// =============================================================================
// 段 4: 远控故障复位（Modbus bCommReset → GVL.bError_Reset）
// =============================================================================
IF NOT GVL.bLocalCtrl THEN
	IF Modbus_Slave.bCommReset THEN
		GVL.bError_Reset:=TRUE;
	ELSE
		GVL.bError_Reset:=FALSE;
	END_IF
END_IF

// =============================================================================
// 段 5: 远控 rPosCmd 赋值
// AGC搜索永久禁用：不跨程序访问Az_ctrl_1/El_ctrl_1内部搜索变量，
// 目标位置只使用ACU指向值及原有测试偏置。
// =============================================================================
Axis1Remote.rPosCmd:=Modbus_Slave.rPosCmd_1 + eltest;
Axis2Remote.rPosCmd:=Modbus_Slave.rPosCmd_2 - aztest;

// =============================================================================
// 段 6: 远控三轴命令参量 — Modbus_Slave → AxisXRemote（核心汇总）
// 俯仰/方位 rPosCmd 已在段 5 赋值；此处写 mode/速度/PID/加减速等
// =============================================================================
Axis1Remote.iMode:=Modbus_Slave.iMode_1;
Axis1Remote.rVelCmd:=Modbus_Slave.rVelcmd_1;
//Axis1Remote.rPosCmd:=Modbus_Slave.rPosCmd_1;
Axis1Remote.rP:=Modbus_Slave.rP_1;
Axis1Remote.rI:=Modbus_Slave.rI_1;
Axis1Remote.rD:=Modbus_Slave.rD_1;
Axis1Remote.rPIDThre:=Modbus_Slave.rThreshold_1;
Axis1Remote.rAcceleration:=Modbus_Slave.rAcceleration_1;
Axis1Remote.rDeceleration:=Modbus_Slave.rDeceleration_1;
Axis1Remote.rVel_Limit:=Modbus_Slave.rVelLim_1;

Axis2Remote.iMode:=Modbus_Slave.iMode_2;
Axis2Remote.rVelCmd:=Modbus_Slave.rVelcmd_2;
//Axis2Remote.rPosCmd:=Modbus_Slave.rPosCmd_2;
Axis2Remote.rP:=Modbus_Slave.rP_2;
Axis2Remote.rI:=Modbus_Slave.rI_2;
Axis2Remote.rD:=Modbus_Slave.rD_2;
Axis2Remote.rPIDThre:=Modbus_Slave.rThreshold_2;
Axis2Remote.rAcceleration:=Modbus_Slave.rAcceleration_2;
Axis2Remote.rDeceleration:=Modbus_Slave.rDeceleration_2;
Axis2Remote.rVel_Limit:=Modbus_Slave.rVelLim_2;

Axis3Remote.iMode:=Modbus_Slave.iMode_3;
Axis3Remote.rVelCmd:=Modbus_Slave.rVelcmd_3;
Axis3Remote.rPosCmd:=Modbus_Slave.rPosCmd_3;
Axis3Remote.rP:=Modbus_Slave.rP_3;
Axis3Remote.rI:=Modbus_Slave.rI_3;
Axis3Remote.rD:=Modbus_Slave.rD_3;
Axis3Remote.rPIDThre:=Modbus_Slave.rThreshold_3;
Axis3Remote.rAcceleration:=Modbus_Slave.rAcceleration_3;
Axis3Remote.rDeceleration:=Modbus_Slave.rDeceleration_3;

(* 本控 HMI 赋值块已注释，本控参数由其他 POU 写入 AxisXLocal *)

// =============================================================================
// 段 7: 俯仰预限位 → Error_code 201(正)/202(负)，置位上报标志
// bPrelimit_1 由硬件 IO 映射；触发后根据当前位置方向区分正负限位
// =============================================================================
IF bPrelimit_1 THEN
	IF rFctPosReal_1>0 THEN
		bPrelimit_Positive_1:=TRUE;
		Error_code:=201;
	ELSE
		bPrelimit_Negative_1:=TRUE;
		Error_code:=202;
	END_IF;
ELSE
	bPrelimit_Negative_1:=FALSE;
	bPrelimit_Positive_1:=FALSE;
	;
END_IF

// =============================================================================
// 段 8: 方位预限位（仅置标志，Error_code 203/204 已注释）
// =============================================================================
IF bPrelimit_2 THEN
	IF rFctPosReal_2>0 THEN
		//Error_code:=203;
		bPrelimit_Positive_2:=TRUE;;
	ELSE
		//Error_code:=204;
		bPrelimit_Negative_2:=TRUE;;
	END_IF;
ELSE
	bPrelimit_Negative_2:=FALSE;
	bPrelimit_Positive_2:=FALSE;
	;
END_IF

// =============================================================================
// 段 9: 扫描找到中心 — 基带锁 且 Eb/N0>rEbN0LockThre 且 √(Ua²+Ue²)<=rErrorThre
// =============================================================================
bBaseLock := (READ_485.recBuf[1] = 1);
bErrorLock  := (SQRT(GVL.Chan1Ua * GVL.Chan1Ua + GVL.Chan1Ue * GVL.Chan1Ue) <= PersistentVars.rErrorThre);
bLockSearch := (READ_485.Chan1EB_N0 > PersistentVars.rEbN0LockThre) AND bBaseLock AND bErrorLock;

// =============================================================================
// 段 10: 启扫 — 双轴程引、未自跟踪、基带锁连续 2s（200×10ms）触发一次
// 扫描过程中允许短暂掉锁；找到合格点后本程引周期内不再重扫
// 退出程引则清锁存
// =============================================================================
IF Axis1Remote.iMode = 4 AND Axis2Remote.iMode = 4 THEN
	IF NOT GVL.bStartTrack AND NOT bScanRun AND NOT bScanFound THEN
		IF bBaseLock THEN
			icountBaseLock := icountBaseLock + 1;
			IF icountBaseLock >= 200 THEN
				icountBaseLock := 200;
				bScanRun := TRUE;
			END_IF
		ELSE
			icountBaseLock := 0;
		END_IF
	ELSE
		icountBaseLock := 0;
	END_IF
	enable := bScanRun AND NOT GVL.bStartTrack;
ELSE
	enable := FALSE;
	bScanRun := FALSE;
	bScanDoneOnce := FALSE;
	bScanFound := FALSE;
	az_latch := 0.0;
	el_latch := 0.0;
	icountBaseLock := 0;
	iTrackStableCount := 0;
END_IF

// 螺旋参数自动适配：界面只需输入波束角和每格周期数。
// 螺距=波束角/4，现有状态机扫描至4*rdelta，即半范围=±1个波束角。
// 完整轨迹固定53个网格段；MainTask为10ms，因此总时间=53*周期数*0.01s。
IF PersistentVars.rBeamWidth > 1.0E-6 THEN
	rdelta := PersistentVars.rBeamWidth / 4.0;
ELSE
	rdelta := 0.25;
END_IF
IF PersistentVars.iSpiralStepCycles >= 1 THEN
	rStep := INT_TO_REAL(PersistentVars.iSpiralStepCycles);
ELSE
	rStep := 12.0;
END_IF
rSpiralFullTimeSec := 53.0 * rStep * 0.01;

// =============================================================================
// 段 11: 调用扫描 FB — 直线或螺旋；locked=找到中心，连续10拍（约100ms）锁存
// =============================================================================
IF NOT PersistentVars.bSearchType THEN
	fbScan(
		enable      := enable,
		delta       := PersistentVars.rAzScanRange,
		delta1      := PersistentVars.rElScanRange,
		n_steps     := PersistentVars.rStep,
		agc      := READ_485.Chan1EB_N0,
		locked     := bLockSearch,
		AzError := AzError,
		elError := elError,
		bTrackState := GVL.bStartTrack,
		azimuth     => az_out,
		elevation   => el_out,
		done        => done_out,
		bType  =>bType_out
	);
ELSE
	fbScan1(
		enable      := enable,
		delta       := rdelta,
		n_steps     := rStep,
		agc      := READ_485.Chan1EB_N0,
		locked     := bLockSearch,
		AzError := AzError,
		elError := elError,
		azimuth     => az_out,
		elevation   => el_out,
		done        => done_out,
		bType  =>bType_out
	);
END_IF

// 扫描结束：找到则锁存 -Error；未找到锁存 0。此后 az_out/el_out 用锁存值
IF bScanRun AND done_out THEN
	bScanRun := FALSE;
	bScanDoneOnce := TRUE;
	IF bType_out THEN
		az_latch := az_out;
		el_latch := el_out;
		bScanFound := TRUE;
	ELSE
		az_latch := 0.0;
		el_latch := 0.0;
		bScanFound := FALSE;
	END_IF
END_IF
IF GVL.bStartTrack AND bScanRun THEN
	bScanRun := FALSE;
	bScanDoneOnce := TRUE;
END_IF
IF NOT bScanRun THEN
	az_out := az_latch;
	el_out := el_latch;
END_IF

// =============================================================================
// 段 12: 两种转跟模式严格隔离
// bTrackingMode=FALSE：ACU转跟，严格跟随HR[7].B10的当前电平。
// bTrackingMode=TRUE ：自动转跟，仅在扫描100ms确认完成(bScanFound)后开启；
//                      基带连续失锁2s后退出，并允许恢复锁定后重新扫描。
// =============================================================================
IF PersistentVars.bTrackingMode <> bTrackingModeLast THEN
	// 切换模式时清除旧模式内部状态，随后完全服从新模式的信号源。
	bAutoTrackLatch := FALSE;
	iBaseLockLostCount := 0;
	IF PersistentVars.bTrackingMode THEN
		// 切入自动模式必须重新完成一次扫描确认，不能沿用ACU模式期间的旧结果。
		bScanFound := FALSE;
		bScanDoneOnce := FALSE;
	END_IF
	bTrackingModeLast := PersistentVars.bTrackingMode;
END_IF

IF NOT PersistentVars.bTrackingMode THEN
	// ACU转跟模式：扫描结果不能开关自跟踪，B10置1/置0立即生效。
	bAutoTrackLatch := FALSE;
	iBaseLockLostCount := 0;
	GVL.bStartTrack := Modbus_Slave.bStartTrackAcu;
ELSE
	// 自动转跟只允许在双轴程引模式工作，完全忽略ACU的B10。
	IF Axis1Remote.iMode = 4 AND Axis2Remote.iMode = 4 THEN
		IF bScanFound THEN
			bAutoTrackLatch := TRUE;
		END_IF

		IF bAutoTrackLatch THEN
			IF bBaseLock THEN
				iBaseLockLostCount := 0;
			ELSE
				IF iBaseLockLostCount < iBaseLockLostNeed THEN
					iBaseLockLostCount := iBaseLockLostCount + 1;
				END_IF
				IF iBaseLockLostCount >= iBaseLockLostNeed THEN
					bAutoTrackLatch := FALSE;
					bScanFound := FALSE;
					bScanDoneOnce := FALSE;
				END_IF
			END_IF
		ELSE
			iBaseLockLostCount := 0;
		END_IF
	ELSE
		bAutoTrackLatch := FALSE;
		iBaseLockLostCount := 0;
	END_IF
	GVL.bStartTrack := bAutoTrackLatch;
END_IF

IF NOT (Axis1Remote.iMode = 4 AND Axis2Remote.iMode = 4) THEN
	icountLostLocking := 0;
	icountLostLocking1 := 0;
	Iconutwait := 0;
END_IF

// =============================================================================
// 段 13: 扫描偏置向自跟踪平滑交接
// 扫描偏置先锁存以进入自跟踪；误差电压稳定进入 1V 内后，才慢速释放扫描偏置。
// 自跟踪偏置在此过程中接管补偿。误差变大时立即暂停释放，保留当前扫描偏置。
// =============================================================================
rTrackErrorMag := SQRT(GVL.Chan1Ua * GVL.Chan1Ua + GVL.Chan1Ue * GVL.Chan1Ue);
IF GVL.bStartTrack AND bScanFound AND bBaseLock AND (rTrackErrorMag <= rTrackReleaseThreshold) THEN
	IF iTrackStableCount < iTrackStableNeed THEN
		iTrackStableCount := iTrackStableCount + 1;
	END_IF
	IF iTrackStableCount >= iTrackStableNeed THEN
		IF az_latch > rScanReleaseStep THEN
			az_latch := az_latch - rScanReleaseStep;
		ELSIF az_latch < -rScanReleaseStep THEN
			az_latch := az_latch + rScanReleaseStep;
		ELSE
			az_latch := 0.0;
		END_IF

		IF el_latch > rScanReleaseStep THEN
			el_latch := el_latch - rScanReleaseStep;
		ELSIF el_latch < -rScanReleaseStep THEN
			el_latch := el_latch + rScanReleaseStep;
		ELSE
			el_latch := 0.0;
		END_IF
	END_IF
ELSE
	iTrackStableCount := 0;
END_IF

// 让本周期后续使用的扫描偏置也反映交接后的值。
IF NOT bScanRun THEN
	az_out := az_latch;
	el_out := el_latch;
END_IF
