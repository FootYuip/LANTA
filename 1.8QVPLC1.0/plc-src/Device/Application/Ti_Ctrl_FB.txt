# kind=functionBlock
# name=Ti_Ctrl_FB
# objectId=ebe00c16-e4b5-485c-a823-5dd11ef20e18

VAR_INPUT
    sLocal : AxisCrtlStruct;
    sCtrl : AxisCrtlStruct;
    sStatus : AxisStatus;
    sVisible : AxisVisible;
    rRealPos : REAL;
    rPosLimK : REAL;
END_VAR

VAR_OUTPUT
    logicError : INT;
END_VAR

VAR
    VelPID_1 : PID;
    VelPID_2 : PID;
    TON_0 : TON;
    bTick : BOOL;
    tr : BOOL;
    icount1 : INT;
    icount2 : INT;
    icount3 : INT;
    bfin1 : BOOL;
    bfin2 : BOOL;
    istate : INT;
    icount : INT;
END_VAR

// --- implementation (ST) ---
// 下电时复位控制字bit8
IF sStatus.Disabled OR(NOT sStatus.Moving)THEN
	bTiCtrlBit8 := FALSE;
END_IF

//	错误复位功能
IF GVL.bError_Reset THEN
	sCtrl.iMode:=6;
	sLocal.iMode:=6;
	GVL.Error_code:=0;
END_IF
IF GVL.Error_code =0 THEN
	GVL.bError_Reset:=FALSE;
END_IF

/////////心跳报错清除/////////
IF GVL.eTiErrorcode = 16#8130 AND NOT tr THEN
	GVL.bTiCtrlBit7 := TRUE;
	tr:=TRUE;
	istate:=0;
	icount:=0;
	// 写入下电控制字
	GVL.bTiCtrlBit3 := FALSE;
	GVL.bTiCtrlBit0 := FALSE;
	GVL.bTiCtrlBit1 := FALSE;
	GVL.bTiCtrlBit2 := FALSE;
END_IF
IF tr THEN
	icount1 := icount1 + 1;
END_IF
IF icount1 > 101 THEN
	icount1 := 0;
	tr := FALSE;
	GVL.bTiCtrlBit7 := FALSE;
	bfin1 := TRUE;
END_IF
IF bfin1 THEN
	icount2 := icount2 + 1;
END_IF
IF icount2 > 101 THEN
	bfin1 := FALSE;
	icount2 := 0;
END_IF
/////////////////////////////

IF GVL.Error_code=0 AND GVL.eTiErrorcode = 0 AND NOT GVL.bTiCtrlBit7  AND NOT tr AND NOT bfin1 AND NOT bfin2 THEN
	IF GVL.bLocalCtrl=FALSE THEN //远控
		IF sCtrl.iMode <> 1 THEN
			istate:=0;
			icount:=0;
		END_IF
		CASE sCtrl.iMode OF
			MODES.powerOff: // 下电
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN // 运动中，先停止运动
						GVL.diTiTargetVel := 0;
						bTiCtrlBit8 := TRUE;
					ELSE
						// 写入下电控制字
						GVL.bTiCtrlBit3 := FALSE;
						GVL.bTiCtrlBit0 := FALSE;
						GVL.bTiCtrlBit1 := FALSE;
						GVL.bTiCtrlBit2 := FALSE;
					END_IF
				END_IF
			MODES.powerOn:  // 上电
				IF sStatus.Disabled THEN
					// 目标速度清零
					GVL.diTiTargetVel := 0;
					IF GVL.siTiMode <> 3 THEN // 上电时模式默认为3，速度模式
						GVL.siTiMode := 3;
					END_IF
					IF siTiMode = 3 THEN
						CASE istate OF
							0:
								GVL.bTiCtrlBit1 := TRUE;
								GVL.bTiCtrlBit2 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=1;
								END_IF
							1:
								GVL.bTiCtrlBit0 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=2;
								END_IF
							2:
								GVL.bTiCtrlBit3 := TRUE;
						END_CASE
					END_IF
				ELSE
					IF sStatus.Moving THEN
						GVL.diTiTargetVel := 0;
						bTiCtrlBit8 := TRUE;
					ELSE
						bTiCtrlBit8 := FALSE;
					END_IF
				END_IF
			MODES.positionMode: // 远控指向模式
				IF NOT sStatus.Disabled THEN
					IF sCtrl.bStartAutoTrack THEN
						(*
						//自跟踪模式下，由误差电压直接驱动
						VelPID_1(rTarget:=sCtrl.rChanU + rRealPos,
								rFeedback:=rRealPos,
								rPosLimPositive:=sCtrl.rSoft_limit_Postive,
								rPosLimNegative:=sCtrl.rSoft_limit_Negative,
								rVelLim:=sCtrl.rVel_Limit,
								rKp:=sCtrl.rP_AutoTrack,//自跟踪PI系数3,0.02
								rKi:=sCtrl.rI_AutoTrack,
								rKd:=sCtrl.rD,
								bPIDEnable:=sCtrl.bPIDEnable,
								iThreshold:= -1,
								rPIDOutput=>sCtrl.rPIDCmdVel);
*)							
					ELSE
						//PID计算速度指令
						VelPID_1(rTarget:=sCtrl.rPosCmd,
							rFeedback:=rRealPos,
							rVelLim:=sCtrl.Pos_Vel,
							rPosLimPositive:=sCtrl.rSoft_limit_Postive,
							rPosLimNegative:=sCtrl.rSoft_limit_Negative,
							rKp:=0.8,//程引PI系数地带：1,0.002  高带：3,0.02
							rKi:=0,
							rKd:=0,
							bPIDEnable:=sCtrl.bPIDEnable,
							iThreshold:=sCtrl.rPIDThre,
							rPIDOutput=>sCtrl.rPIDCmdVel);
					END_IF
					GVL.diTiTargetVel := REAL_TO_DINT(sCtrl.rPIDCmdVel * 524288 / 360);
					GVL.diTiTargetAcc := REAL_TO_UDINT(10 * 524288 / 360);   //加速度
					GVL.diTiTargetDec := REAL_TO_UDINT(10 * 524288 / 360);   //减速度
				END_IF
			MODES.velocityMode://手控速度模式-远控模式
				IF NOT sStatus.Disabled THEN
					IF ABS(sCtrl.rVelCmd) <= sCtrl.rVel_Limit THEN
						GVL.diTiTargetVel := REAL_TO_DINT(sCtrl.rVelCmd * 524288 / 360);
					ELSIF sCtrl.rVelCmd > sCtrl.rVel_Limit THEN
						GVL.diTiTargetVel := REAL_TO_DINT(sCtrl.rVel_Limit * 524288 / 360);
					ELSE
						GVL.diTiTargetVel := REAL_TO_DINT(-sCtrl.rVel_Limit * 524288 / 360);
					END_IF
					GVL.diTiTargetAcc := REAL_TO_UDINT(sCtrl.rAcceleration * 524288 / 360);   //加速度
					GVL.diTiTargetDec := REAL_TO_UDINT(sCtrl.rDeceleration * 524288 / 360);   //减速度
				END_IF
			MODES.programBootMode://PID速度模式-正常模式 程引
				IF NOT sStatus.Disabled THEN
					IF sCtrl.bStartAutoTrack THEN
						//自跟踪模式下，由误差电压直接驱动
						VelPID_1(rTarget:=sCtrl.rChanU + rRealPos,
								rFeedback:=rRealPos,
								rPosLimPositive:=sCtrl.rSoft_limit_Postive,
								rPosLimNegative:=sCtrl.rSoft_limit_Negative,
								rVelLim:=sCtrl.rVel_Limit,
								rKp:=sCtrl.rP_AutoTrack,//自跟踪PI系数3,0.02
								rKi:=sCtrl.rI_AutoTrack,
								rKd:=sCtrl.rD,
								bPIDEnable:=sCtrl.bPIDEnable,
								iThreshold:= -1,
								rPIDOutput=>sCtrl.rPIDCmdVel);							
					ELSE
						//PID计算速度指令
						VelPID_1(rTarget:=sCtrl.rPIDCmdPos,
							rFeedback:=rRealPos,
							rVelLim:=sCtrl.rVel_Limit,
							rPosLimPositive:=sCtrl.rSoft_limit_Postive,
							rPosLimNegative:=sCtrl.rSoft_limit_Negative,
							rKp:=sCtrl.rP,//程引PI系数地带：1,0.002  高带：3,0.02
							rKi:=sCtrl.rI,
							rKd:=sCtrl.rD,
							bPIDEnable:=sCtrl.bPIDEnable,
							iThreshold:=sCtrl.rPIDThre,
							rPIDOutput=>sCtrl.rPIDCmdVel);
					END_IF
					GVL.diTiTargetVel := REAL_TO_DINT(sCtrl.rPIDCmdVel * 524288 / 360);
					GVL.diTiTargetAcc := REAL_TO_UDINT(sCtrl.rAcceleration * 524288 / 360);   //加速度
					GVL.diTiTargetDec := REAL_TO_UDINT(sCtrl.rDeceleration * 524288 / 360);   //减速度
				END_IF
			MODES.reset://重置
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN
						GVL.diTiTargetVel := 0;
						bTiCtrlBit8 := TRUE;
					ELSE
						bTiCtrlBit8 := FALSE;
					END_IF
				END_IF
			MODES.halt://待机
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN
						GVL.diTiTargetVel := 0;
						bTiCtrlBit8 := TRUE;
					ELSE
						bTiCtrlBit8 := FALSE;
					END_IF
				END_IF
			
		END_CASE
	ELSE // 本控
		IF sLocal.iMode <> 1 THEN
			istate:=0;
			icount:=0;
		END_IF
		CASE sLocal.iMode OF
			MODES.powerOff: // 下电
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN // 运动中，先停止运动
						GVL.diTiTargetVel := 0;
						bTiCtrlBit8 := TRUE;
					ELSE
						// 写入下电控制字
						GVL.bTiCtrlBit3 := FALSE;
						GVL.bTiCtrlBit0 := FALSE;
						GVL.bTiCtrlBit1 := FALSE;
						GVL.bTiCtrlBit2 := FALSE;
					END_IF
				END_IF
			MODES.powerOn:  // 上电
				IF sStatus.Disabled THEN
					// 目标速度清零
					GVL.diTiTargetVel := 0;
					IF GVL.siTiMode <> 3 THEN // 上电时模式默认为3，速度模式
						GVL.siTiMode := 3;
					END_IF
					IF siTiMode = 3 THEN
						CASE istate OF
							0:
								GVL.bTiCtrlBit1 := TRUE;
								GVL.bTiCtrlBit2 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=1;
								END_IF
							1:
								GVL.bTiCtrlBit0 := TRUE;
								icount := icount + 1;
								IF icount = 20 THEN
									icount:=0;
									istate:=2;
								END_IF
							2:
								GVL.bTiCtrlBit3 := TRUE;
						END_CASE
					END_IF
				ELSE
					IF sStatus.Moving THEN
						GVL.diTiTargetVel := 0;
						bTiCtrlBit8 := TRUE;
					ELSE
						bTiCtrlBit8 := FALSE;
					END_IF
				END_IF
			MODES.velocityMode://手控速度模式-本控模式
				IF NOT sStatus.Disabled THEN
					IF ABS(sLocal.rVelCmd) <= sLocal.rVel_Limit THEN
						GVL.diTiTargetVel := REAL_TO_DINT(sLocal.rVelCmd * 524288 / 360);
					ELSIF sLocal.rVelCmd > sLocal.rVel_Limit THEN
						GVL.diTiTargetVel := REAL_TO_DINT(sLocal.rVel_Limit * 524288 / 360);
					ELSE
						GVL.diTiTargetVel := REAL_TO_DINT(-sLocal.rVel_Limit * 524288 / 360);
					END_IF
					GVL.diTiTargetAcc := REAL_TO_UDINT(sLocal.rAcceleration * 524288 / 360);   //加速度
					GVL.diTiTargetDec := REAL_TO_UDINT(sLocal.rDeceleration * 524288 / 360);   //减速度
				END_IF
			MODES.programBootMode://本控位置模式
				IF NOT sStatus.Disabled THEN
					IF NOT sLocal.bPIDEnable THEN
						GVL.diTiTargetVel := 0;							
					ELSE
						//PID计算速度指令
						 VelPID_2(rTarget:=sLocal.rPosCmd,
							     rFeedback:=rRealPos,
								 rPosLimPositive:=sLocal.rSoft_limit_Postive,//软限位
								 rPosLimNegative:=sLocal.rSoft_limit_Negative,
								 rKp:=sLocal.rP,
								 rKi:=sLocal.rI,
								 rKd:=sLocal.rD,
								 iThreshold:=sLocal.rPIDThre,
								 bPIDEnable:=sLocal.bPIDEnable,
								 rVelLim:=sLocal.Pos_Vel,
								 rPIDOutput=>sLocal.rPIDCmdVel);//普通PID
					END_IF
					GVL.diTiTargetVel := REAL_TO_DINT(sLocal.rPIDCmdVel * 524288 / 360);
					GVL.diTiTargetAcc := REAL_TO_UDINT(sLocal.rAcceleration * 524288 / 360);   //加速度
					GVL.diTiTargetDec := REAL_TO_UDINT(sLocal.rDeceleration * 524288 / 360);   //减速度
				END_IF
			MODES.reset://重置
			// 是否需要
			MODES.halt://待机
				IF NOT sStatus.Disabled THEN
					IF sStatus.Moving THEN
						GVL.diTiTargetVel := 0;
						bTiCtrlBit8 := TRUE;
					ELSE
						bTiCtrlBit8 := FALSE;
					END_IF
				END_IF
			
		END_CASE
	END_IF
ELSIF GVL.Error_code<>0 AND GVL.eTiErrorcode = 0 AND NOT GVL.bTiCtrlBit7  AND NOT tr AND NOT bfin1 AND NOT bfin2 THEN
	// 出现错误，轴停止运动
	GVL.diTiTargetVel := 0;
	bTiCtrlBit8 := TRUE;
	TON_0(IN:=bTiCtrlBit8,PT:=T#1S,Q=>,ET=>);//判断stop0执行是否超时,若超时强制下电（手动复位）
	IF TON_0.Q THEN
		bTick:=TRUE;
	END_IF
	IF bTick AND sStatus.Moving THEN // 超时强制下电
		// 写入下电控制字
		GVL.bTiCtrlBit3 := FALSE;
		GVL.bTiCtrlBit0 := FALSE;
		GVL.bTiCtrlBit1 := FALSE;
		GVL.bTiCtrlBit2 := FALSE;
	END_IF
END_IF
